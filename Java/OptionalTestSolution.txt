package com.epam.dsa;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.stream.Collectors;

class Chapter {
    String title;
    String content;
    String summary;

    public Chapter(String title, String content, String summary) {
        this.title = title;
        this.content = content;
        this.summary = summary;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public String getContent() {
        return content;
    }

    public void setContent(String content) {
        this.content = content;
    }

    public String getSummary() {
        return Optional.ofNullable(this.summary).map(String::toUpperCase).orElse("NO SUMMARY PRESENT");
    }

    public void setSummary(String summary) {
        this.summary = summary;
    }

}

class Book {
    String title;
    List<Chapter> chapters;

    public Book(String title, List<Chapter> chapters) {
        this.title = title;
        this.chapters = chapters;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public List<Chapter> getChapters() {
        return chapters;
    }

    public void setChapters(List<Chapter> chapters) {
        this.chapters = chapters;
    }
    public Optional<List<String>> getAllSummaries(Book book) {
        return Optional.of(Optional.ofNullable(book)
                .map(Book::getChapters)
                .stream()
                .flatMap(List::stream)
                .map(Chapter::getSummary)
                .toList());
    }
}
